const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId(); const PASSCODE = "123456"; // Ganti password admin Anda di sini function doGet(e) { const action = e.parameter.action; const guestId = e.parameter.guestId; if (action === "getGuest") { return jsonResponse(getGuestData(guestId)); } else if (action === "getWishes") { return jsonResponse(getWishesData()); } else if (action === "getAdminData") { const pass = e.parameter.passcode; if (pass !== PASSCODE) return jsonResponse({ error: "Unauthorized" }); return jsonResponse(getAdminDashboardData()); } return jsonResponse({ status: "active" }); } function doPost(e) { try { const contents = JSON.parse(e.postData.contents); const action = contents.action; if (action === "submitRSVP") { return jsonResponse(saveRSVP(contents.data)); } else if (action === "submitWish") { return jsonResponse(saveWish(contents.data)); } else if (action === "adminUpdateGuest") { if (contents.passcode !== PASSCODE) return jsonResponse({ error: "Unauthorized" }); return jsonResponse(adminUpdateGuest(contents.data)); } else if (action === "adminAddGuest") { if (contents.passcode !== PASSCODE) return jsonResponse({ error: "Unauthorized" }); return jsonResponse(adminAddGuest(contents.data)); } } catch (err) { return jsonResponse({ error: err.toString() }); } } function jsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function getGuestData(guestId) { if (!guestId) return { name: "Tamu Undangan", maxGuests: 2 }; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Guests"); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0].toString().trim() === guestId.toString().trim()) { return { guestId: data[i][0], name: data[i][1], phone: data[i][2], category: data[i][3], maxGuests: data[i][4], status: data[i][5], attendingCount: data[i][6], notes: data[i][7] }; } } return { name: "Tamu Undangan", maxGuests: 2 }; } function saveRSVP(data) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Guests"); const rows = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < rows.length; i++) { if (rows[i][0].toString().trim() === data.guestId.toString().trim()) { rowIndex = i + 1; break; } } if (rowIndex !== -1) { sheet.getRange(rowIndex, 6).setValue(data.status); sheet.getRange(rowIndex, 7).setValue(data.attendingCount); sheet.getRange(rowIndex, 8).setValue(data.notes || ""); } const rsvpSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("RSVP_Log"); rsvpSheet.appendRow([ new Date(), data.guestId, data.name, data.phone, data.status, data.attendingCount, data.notes ]); return { success: true }; } function saveWish(data) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Wishes"); sheet.appendRow([new Date(), data.name, data.message]); return { success: true }; } function getWishesData() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Wishes"); const data = sheet.getDataRange().getValues(); const wishes = []; for (let i = data.length - 1; i >= 1; i--) { wishes.push({ timestamp: data[i][0], name: data[i][1], message: data[i][2] }); } return { wishes: wishes }; } function getAdminDashboardData() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Guests"); const data = sheet.getDataRange().getValues(); let totalGuests = 0; let totalInvited = 0; let totalAttending = 0; let totalAbsent = 0; let totalUnconfirmed = 0; let totalPeopleAttending = 0; const list = []; for (let i = 1; i < data.length; i++) { if(!data[i][0]) continue; totalGuests++; const max = parseInt(data[i][4]) || 1; const status = data[i][5] || "Belum Konfirmasi"; const count = parseInt(data[i][6]) || 0; totalInvited += max; if (status === "Hadir") { totalAttending++; totalPeopleAttending += count; } else if (status === "Tidak Hadir") { totalAbsent++; } else { totalUnconfirmed++; } list.push({ guestId: data[i][0], name: data[i][1], phone: data[i][2], category: data[i][3], maxGuests: max, status: status, attendingCount: count, notes: data[i][7] }); } return { summary: { totalGuests, totalInvited, totalAttending, totalAbsent, totalUnconfirmed, totalPeopleAttending }, guests: list }; } function adminUpdateGuest(data) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Guests"); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0].toString().trim() === data.guestId.toString().trim()) { const row = i + 1; sheet.getRange(row, 2).setValue(data.name); sheet.getRange(row, 3).setValue(data.phone); sheet.getRange(row, 4).setValue(data.category); sheet.getRange(row, 5).setValue(data.maxGuests); sheet.getRange(row, 6).setValue(data.status); sheet.getRange(row, 7).setValue(data.attendingCount); sheet.getRange(row, 8).setValue(data.notes); return { success: true }; } } return { success: false, error: "Guest not found" }; } function adminAddGuest(data) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Guests"); sheet.appendRow([ data.guestId, data.name, data.phone, data.category, data.maxGuests, "Belum Konfirmasi", 0, "" ]); return { success: true }; }
